The following stand for all the abbreviations and acronyms used throughout the documentation.

\begin{itemize}
	\item \textbf{VLM}: Vision Language Model
	\item \textbf{LLM}: Large Language Model
	\item \textbf{LMMs}: Large Multimodal Models
	\item \textbf{SOTA}: State-of-the-Art
	\item \textbf{RL}: Reinforcement Learning
	\item \textbf{VLA}: Vision Language Action Model
	\item \textbf{CLIP}: Contrastive Language Image Pretraining
	\item \textbf{ViT}: Vision Transformer
	\item \textbf{CNN}: Convolutional Neural Network
	\item \textbf{RNN}: Recurrent Neural Network
	\item \textbf{PPO}: Proximal Policy Optimization
	\item \textbf{MEDE}: Mean Euclidean Distance Error
	\item \textbf{CV}: Computer Vision
	\item \textbf{USD}: Universal Scene Description
	\item \textbf{JSON}: JavaScript Object Notation
	\item \textbf{GPU}: Graphics Processing Unit
	\item \textbf{DL}: Deep Learning
	\item \textbf{MARL}: Multi Agent Reinforcement Learning

\end{itemize}
